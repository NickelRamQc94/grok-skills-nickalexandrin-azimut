#!/usr/bin/env python3
"""
Indexation et Catégorisation Automatique - Écosystème Nickel
Cible : Tout-les-liens-et-Contenus- & OoSK-vs-OpenAI
Auteur : Junior Willow Nickel Réjean Gemini David Grok Meta DeepSeek PinnochIA Owen Grenier

import json
import urllib.request
import os

# Configuration du dépôt cible
REPO_OWNER = "NickelRamQc94"
REPO_NAME = "Tout-les-liens-et-Contenus-"  # Modifiable pour "OoSK-vs-OpenAI"
MAX_SIZE_BYTES = 25 * 1024 * 1024  # Seuil de 25 MB

CATEGORIES = {
    "🎵 Audio & Musique (MP3/M4A)": [".mp3", ".wav", ".m4a", ".flac", ".ogg"],
    "🎬 Vidéo & Média (MP4/MOV)": [".mp4", ".mov", ".avi", ".mkv", ".webm"],
    "📄 Documents & PDF": [".pdf", ".docx", ".doc", ".txt", ".rtf"],
    "💻 Code & Scripts": [".py", ".ipynb", ".sh", ".rs", ".js", ".ts", ".c"],
    "🧠 Théorie & Structures": [".md", ".json", ".yaml", ".yml", ".csv", ".html"]
}

def fetch_github_tree(owner, repo):
    """Récupère l'arborescence complète du dépôt via l'API GitHub."""
    for branch in ["main", "master"]:
        url = f"https://api.github.com/repos/{owner}/{repo}/git/trees/{branch}?recursive=1"
        req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
        try:
            with urllib.request.urlopen(req) as response:
                data = json.loads(response.read().decode())
                return data.get("tree", []), branch
        except Exception:
            continue
    return [], "main"

def categorize_repo_files(tree, branch):
    """Classe les fichiers par catégorie et vérifie leur taille."""
    categorized = {cat: [] for cat in CATEGORIES}
    categorized["📦 Autres Fichiers"] = []
    
    total_count = 0
    total_size = 0

    for item in tree:
        if item.get("type") == "blob":
            path = item.get("path", "")
            size = item.get("size", 0)
            ext = os.path.splitext(path)[1].lower()
            
            file_info = {
                "path": path,
                "size_mb": round(size / (1024 * 1024), 2) if size else 0,
                "url": f"https://github.com/{REPO_OWNER}/{REPO_NAME}/blob/{branch}/{path}",
                "raw_url": f"https://raw.githubusercontent.com/{REPO_OWNER}/{REPO_NAME}/{branch}/{path}"
            }

            matched = False
            for cat_name, extensions in CATEGORIES.items():
                if ext in extensions:
                    categorized[cat_name].append(file_info)
                    matched = True
                    break
            
            if not matched:
                categorized["📦 Autres Fichiers"].append(file_info)
            
            total_count += 1
            total_size += size

    return categorized, total_count, total_size

def generate_markdown_catalog(categorized, total_count, total_size):
    """Génère le rapport structuré en Markdown."""
    md = []
    md.append(f"# 📊 Indexation & Banque de Données : `{REPO_OWNER}/{REPO_NAME}`")
    md.append(f"**Statut :** Locké & Actif | **Fichiers <= 25 Mo**")
    md.append(f"**Total Fichiers :** {total_count} | **Volume Total :** {round(total_size / (1024 * 1024), 2)} Mo\n")
    md.append("---")

    for cat_name, files in categorized.items():
        if files:
            md.append(f"\n### {cat_name} ({len(files)} élément(s))")
            md.append("| Nom du Fichier / Chemin | Taille (Mo) | Lien GitHub | Lien Direct (Raw) |")
            md.append("| :--- | :---: | :---: | :---: |")
            for f in files:
                md.append(f"| `{f['path']}` | **{f['size_mb']} Mo** | [Consulter]({f['url']}) | [Télécharger]({f['raw_url']}) |")

    return "\n".join(md)

if __name__ == "__main__":
    print(f"🔍 Analyse du dépôt {REPO_OWNER}/{REPO_NAME}...")
    tree, branch = fetch_github_tree(REPO_OWNER, REPO_NAME)
    
    if tree:
        categorized, count, size = categorize_repo_files(tree, branch)
        catalog_md = generate_markdown_catalog(categorized, count, size)
        
        filename = f"INDEX_{REPO_NAME}.md"
        with open(filename, "w", encoding="utf-8") as f:
            f.write(catalog_md)
        print(f"✅ Catalogue généré avec succès dans : {filename}")
    else:
        print("❌ Impossible de lire l'arborescence. Vérifiez l'accès au dépôt.")
